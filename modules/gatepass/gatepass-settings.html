⚙ Gatepass PDF Settings

Configure address and contact numbers for each branch — they appear on every gate pass PDF.

📋 First-time setup — run this SQL in Supabase if the table does not exist yet 
CREATE TABLE IF NOT EXISTS gatepass_branch_settings (
  id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  branch_name text UNIQUE NOT NULL,
  address text,
  chief_security_phone text,
  plant_manager_phone  text,
  updated_at timestamptz DEFAULT now()
);
ALTER TABLE gatepass_branch_settings ENABLE ROW LEVEL SECURITY;
CREATE POLICY "read"  ON gatepass_branch_settings FOR SELECT TO authenticated USING (true);
CREATE POLICY "write" ON gatepass_branch_settings FOR ALL    TO authenticated USING (true) WITH CHECK (true);
